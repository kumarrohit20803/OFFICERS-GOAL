      Supabase SQL Setup (Run once in SQL Editor) 
      create table if not exists habits (
  id text primary key,
  officer text not null,
  name text not null,
  created_at timestamptz default now()
);

create table if not exists habit_logs (
  id text primary key,
  officer text not null,
  habit_id text not null,
  log_date text not null
);

create table if not exists daily_goals (
  id text primary key,
  officer text not null,
  goal_date text not null,
  title text not null,
  priority text not null default 'important',
  completed boolean not null default false
);

alter table habits disable row level security;
alter table habit_logs disable row level security;
alter table daily_goals disable row level security;

alter publication supabase_realtime add table habits;
alter publication supabase_realtime add table habit_logs;
create table if not exists mock_tests (
  id text primary key,
  officer text not null,
  exam_name text not null default 'SSC CGL',
  test_date text not null,
  stage text not null,
  test_type text not null,
  subject text,
  math_marks numeric,
  reasoning_marks numeric,
  english_marks numeric,
  gkgs_marks numeric,
  scored_marks numeric not null,
  total_marks numeric not null,
  cutoff_marks numeric,
  computer_qualified boolean,
  created_at timestamptz default now()
);

alter table mock_tests disable row level security;

alter publication supabase_realtime add table daily_goals;
create table if not exists exam_dates (
  id text primary key,
  exam_name text not null,
  exam_date text not null,
  officer text not null,
  created_at timestamptz default now()
);

alter table exam_dates disable row level security;

alter publication supabase_realtime add table mock_tests;
create table if not exists streak_reasons (
  id text primary key,
  reason_date text not null,
  who_missed text not null,
  reason text not null,
  created_at timestamptz default now()
);

alter table streak_reasons disable row level security;
